/**
 * =========================================================================
 * BACKEND GOOGLE APPS SCRIPT · CAMPUS DE NAVIDAD ALMERÍA 2026
 * =========================================================================
 * Maneja recepción de inscripciones, registro ordenado en Google Sheets sin
 * variables duplicadas y envío de confirmaciones automáticas por correo.
 */

// ID de la Hoja de Cálculo de Google Sheets
var SPREADSHEET_ID = '10bRWNhf4wm98doXomWTU5XMYcQu6FKUfGa-9eWzGJeE';

function doPost(e) {
  var lock = LockService.getScriptLock();
  // Esperar hasta 30 segundos para evitar bloqueos concurrentes
  lock.tryLock(30000);

  try {
    var ss;
    try {
      ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    } catch (errOpen) {
      // Si el script está abierto directamente desde la hoja de cálculo (Extensiones > Apps Script)
      ss = SpreadsheetApp.getActiveSpreadsheet();
      if (!ss) throw errOpen;
    }
    // Usar la hoja 'Inscripciones' si existe, o la primera hoja activa
    var sheet = ss.getSheetByName('Inscripciones') || ss.getActiveSheet();
    var timestamp = new Date();

    // Helper para normalizar parámetros que pueden venir como arrays o valores únicos
    function normalizeArray(name) {
      if (e.parameters && e.parameters[name] && e.parameters[name].length) {
        return e.parameters[name];
      }
      var value = e.parameter ? e.parameter[name] : null;
      if (!value) return [];
      return Array.isArray(value) ? value : [value];
    }

    // 1. Participantes (Arrays)
    var nombres = normalizeArray('participante_nombre[]');
    var nacimientos = normalizeArray('participante_nacimiento[]');
    var grupos = normalizeArray('participante_grupo[]');
    var adultosAcompanantes = normalizeArray('participante_adulto_acompanante[]');

    // 2. Datos del Tutor / Representante Legal
    var tutorNombre = e.parameter.tutor_nombre || '';
    var tutorDni = e.parameter.tutor_dni || '';
    var tutorTelefono = e.parameter.tutor_telefono || '';
    var tutorEmail = e.parameter.tutor_email || '';

    // 3. Camiseta Oficial de Recuerdo
    var camisetaOficial = e.parameter.camiseta || 'No';
    var tallaCamiseta = (camisetaOficial === 'Sí') ? (e.parameter.talla_camiseta || 'Sin especificar') : 'No solicitada';

    // 4. Logística, Salud y Talleres
    var medicasObservaciones = e.parameter.medicas_observaciones || 'Ninguna';
    var tallerPropuesto = e.parameter.taller_propuesto || 'Ninguno';

    // 5. Espacio de Voluntariado
    var voluntarioActivo = e.parameter.voluntario_activo || 'No';
    var voluntarioDias = (voluntarioActivo === 'Sí') ? (e.parameter.voluntario_dias || 'Todos los días') : 'No participa';
    var voluntarioDatos = (voluntarioActivo === 'Sí') ? (e.parameter.voluntario_datos || 'No especificado') : '-';

    // 6. Economía / Aportaciones (¡Una sola vez por inscripción!)
    var costeMaterial = e.parameter.coste_material || '0€';
    var donacionVoluntaria = e.parameter.donacion_voluntaria || '0€';
    var totalEstimado = e.parameter.total_estimado || '0€';
    var metodoPago = e.parameter.metodo_pago || 'No especificado';

    // 7. Acuerdos y Autorizaciones
    var compromisoConvivencia = e.parameter.compromiso_aceptado || 'No';
    var compromisoRopaTalleres = e.parameter.llevar_camiseta_blanca || 'No';
    var autorizaImagen = e.parameter.autoriza_imagen || 'No';
    var autorizacionParticipacion = e.parameter.autorizacion_participacion || e.parameter.autorizacion_medica || 'No';
    var acompanamientoInfantil = e.parameter.acompanamiento_infantil_aceptado || 'No aplica';

    // 8. Filtro Anti-Spam (Honeypot)
    if (e.parameter.website_url_check && e.parameter.website_url_check.trim() !== '') {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        message: 'Spam detectado.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 9. Verificación de Seguridad Anti-Bot
    var captchaVerificado = e.parameter.captcha_verificado || 'No';
    if (captchaVerificado !== 'Sí') {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        message: 'No se ha superado la verificación de seguridad anti-bot.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Validaciones obligatorias de campos clave
    if (!tutorNombre || !tutorDni || !tutorTelefono || !tutorEmail || compromisoConvivencia !== 'Sí' || autorizacionParticipacion !== 'Sí') {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        message: 'Faltan campos obligatorios del tutor o autorizaciones reglamentarias.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (nombres.length === 0 || nacimientos.length === 0 || grupos.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        message: 'Debe haber al menos un participante registrado.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Generar Localizador Único de Inscripción (Ej: NAV-26-8412)
    var randomSuffix = Math.floor(1000 + Math.random() * 9000);
    var localizador = 'NAV-26-' + randomSuffix;

    // Crear encabezados limpios y sin redundancias si la hoja está vacía
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Timestamp',
        'Localizador',
        'Nº Participantes',
        'Listado de Participantes (Nombre, Nacimiento, Grupo)',
        'Adulto Acompañante 5h (Etapa Infantil)',
        'Tutor Nombre Completo',
        'Tutor DNI / NIE',
        'Tutor Teléfono',
        'Tutor Correo Electrónico',
        'Camiseta Recuerdo (Sí/No)',
        'Talla Camiseta',
        'Observaciones Médicas / Alergias',
        'Taller o Talento Propuesto',
        'Colabora como Voluntario',
        'Días Disponibles Voluntario',
        'Detalles / Preferencias Voluntario',
        'Coste Material (5€/part)',
        'Donación Voluntaria',
        'Total Aporte Estimado',
        'Método de Pago',
        'Compromiso Convivencia',
        'Compromiso Ropa Talleres',
        'Autorización de Imagen',
        'Autorización de Participación'
      ]);
    }

    // Formatear la lista de participantes en un solo bloque legible sin duplicar filas
    var listaParticipantesFormateada = [];
    var listaAcompanantesInfantil = [];

    for (var i = 0; i < nombres.length; i++) {
      var itemPart = (i + 1) + '. ' + nombres[i] + ' (Nac: ' + nacimientos[i] + ' - ' + grupos[i] + ')';
      listaParticipantesFormateada.push(itemPart);

      if (adultosAcompanantes && adultosAcompanantes[i] && adultosAcompanantes[i] !== 'No aplica' && adultosAcompanantes[i].trim() !== '') {
        listaAcompanantesInfantil.push(nombres[i] + ': ' + adultosAcompanantes[i]);
      }
    }

    var textoParticipantes = listaParticipantesFormateada.join('\n');
    var textoAcompanantes = listaAcompanantesInfantil.length > 0 ? listaAcompanantesInfantil.join('\n') : 'No aplica';

    // REGISTRO LIMPIO: UNA SOLA FILA POR INSCRIPCIÓN (Cero datos duplicados)
    sheet.appendRow([
      timestamp,
      localizador,
      nombres.length,
      textoParticipantes,
      textoAcompanantes,
      tutorNombre,
      tutorDni,
      tutorTelefono,
      tutorEmail,
      camisetaOficial,
      tallaCamiseta,
      medicasObservaciones,
      tallerPropuesto,
      voluntarioActivo,
      voluntarioDias,
      voluntarioDatos,
      costeMaterial,
      donacionVoluntaria,
      totalEstimado,
      metodoPago,
      compromisoConvivencia,
      compromisoRopaTalleres,
      autorizaImagen,
      autorizacionParticipacion
    ]);

    // Opcional: Si en la hoja existe una pestaña llamada 'Participantes', guardar cada niño individualmente para pasar lista
    var sheetParticipantes = ss.getSheetByName('Participantes');
    if (sheetParticipantes) {
      if (sheetParticipantes.getLastRow() === 0) {
        sheetParticipantes.appendRow([
          'Localizador',
          'Nombre del Participante',
          'Fecha Nacimiento',
          'Grupo Asignado',
          'Adulto Acompañante (Infantil 5h)',
          'Tutor Contacto',
          'Teléfono Urgencias',
          'Observaciones Médicas / Alergias'
        ]);
      }
      for (var j = 0; j < nombres.length; j++) {
        var acomp = (adultosAcompanantes && adultosAcompanantes[j]) ? adultosAcompanantes[j] : 'No aplica';
        sheetParticipantes.appendRow([
          localizador,
          nombres[j],
          nacimientos[j],
          grupos[j],
          acomp,
          tutorNombre,
          tutorTelefono,
          medicasObservaciones
        ]);
      }
    }

    // Enviar correo de confirmación al tutor
    enviarCorreoConfirmacion(
      localizador,
      tutorNombre,
      tutorEmail,
      nombres,
      nacimientos,
      grupos,
      adultosAcompanantes,
      costeMaterial,
      donacionVoluntaria,
      totalEstimado,
      metodoPago,
      voluntarioActivo,
      voluntarioDias
    );

    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      localizador: localizador,
      message: '¡Inscripción registrada con éxito! Tu localizador es: ' + localizador
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    Logger.log('Error en doPost: ' + error.toString());
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      message: 'Error en el servidor: ' + error.toString()
    })).setMimeType(ContentService.MimeType.JSON);

  } finally {
    lock.releaseLock();
  }
}

/**
 * Función para comprobar los nombres de las columnas en tu hoja de Google Sheets
 */
function testVerificarEncabezados() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheet = ss.getSheetByName('Inscripciones') || ss.getActiveSheet();

  if (sheet.getLastRow() === 0) {
    Logger.log('[INFO] La hoja está vacía. Ejecuta testDoPost() primero para inicializar encabezados.');
    return;
  }

  var encabezados = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  Logger.log('[INFO] Encabezados actuales en Google Sheets (' + encabezados.length + ' columnas):');
  for (var i = 0; i < encabezados.length; i++) {
    Logger.log('Columna ' + (i + 1) + ': ' + encabezados[i]);
  }
}
